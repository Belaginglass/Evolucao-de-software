package com.tcc.iterativa.model;

import jakarta.persistence.*;
import java.time.LocalDate;
import java.time.LocalTime;
import java.time.LocalDateTime;

@Entity
@Table(name = "apadrinhamentos")
public class Apadrinhamento {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne
    @JoinColumn(name = "animal_id", nullable = false)
    private Animal animal;

    @ManyToOne
    @JoinColumn(name = "adotante_id", nullable = false)
    private Usuario adotante;

    @Column(nullable = false)
    private String tipoServico;

    private LocalDate dia;
    private LocalTime horario;

    @Column(nullable = false)
    private String status; // SOLICITADO, CONFIRMADO

    private LocalDateTime dataPedido;

    public Apadrinhamento() {}

    public Apadrinhamento(Animal animal, Usuario adotante, String tipoServico, LocalDate dia, LocalTime horario, String status, LocalDateTime dataPedido) {
        this.animal = animal;
        this.adotante = adotante;
        this.tipoServico = tipoServico;
        this.dia = dia;
        this.horario = horario;
        this.status = status;
        this.dataPedido = dataPedido;
    }

    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }

    public Animal getAnimal() { return animal; }
    public void setAnimal(Animal animal) { this.animal = animal; }

    public Usuario getAdotante() { return adotante; }
    public void setAdotante(Usuario adotante) { this.adotante = adotante; }

    public String getTipoServico() { return tipoServico; }
    public void setTipoServico(String tipoServico) { this.tipoServico = tipoServico; }

    public LocalDate getDia() { return dia; }
    public void setDia(LocalDate dia) { this.dia = dia; }

    public LocalTime getHorario() { return horario; }
    public void setHorario(LocalTime horario) { this.horario = horario; }

    public String getStatus() { return status; }
    public void setStatus(String status) { this.status = status; }

    public LocalDateTime getDataPedido() { return dataPedido; }
    public void setDataPedido(LocalDateTime dataPedido) { this.dataPedido = dataPedido; }
}
