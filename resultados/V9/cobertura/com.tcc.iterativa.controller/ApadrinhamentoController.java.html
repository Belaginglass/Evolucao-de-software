package com.tcc.iterativa.controller;

import com.tcc.iterativa.model.Apadrinhamento;
import com.tcc.iterativa.model.DoacaoFinanceira;
import com.tcc.iterativa.model.Animal;
import com.tcc.iterativa.service.AdocaoService;
import com.tcc.iterativa.service.ApadrinhamentoService;
import org.springframework.security.core.Authentication;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;

import java.time.LocalDate;
import java.time.LocalTime;
import java.util.Optional;

@Controller
public class ApadrinhamentoController {

    private final ApadrinhamentoService apadrinhamentoService;
    private final AdocaoService adocaoService;

    public ApadrinhamentoController(ApadrinhamentoService apadrinhamentoService, AdocaoService adocaoService) {
        this.apadrinhamentoService = apadrinhamentoService;
        this.adocaoService = adocaoService;
    }

    @GetMapping("/apadrinhamento/causa/{id}")
    public String apadrinhamentoCausaExterna(@PathVariable Long id) {
        // UC007: Direciona para link externo cadastrado
        return "redirect:https://example.com/projeto-causa-" + id;
    }

    @GetMapping("/apadrinhamento/animal/{id}")
    public String apadrinhamentoAnimalForm(@PathVariable Long id, Model model, Authentication authentication) {
        Optional<Animal> animalOpt = adocaoService.buscarAnimalPorId(id);
        if (animalOpt.isEmpty()) {
            return "redirect:/adote-aqui";
        }
        model.addAttribute("animal", animalOpt.get());
        model.addAttribute("autenticado", authentication != null);
        return "apadrinhamento/animal";
    }

    @PostMapping("/apadrinhamento/animal")
    public String solicitarApadrinhamentoAnimal(
            @RequestParam Long animalId,
            @RequestParam String tipoServico,
            @RequestParam LocalDate dia,
            @RequestParam LocalTime horario,
            Authentication authentication,
            Model model) {
        if (authentication == null) {
            return "redirect:/login?redirect=/apadrinhamento/animal/" + animalId;
        }
        try {
            apadrinhamentoService.criarApadrinhamentoServico(animalId, authentication.getName(), tipoServico, dia, horario);
            model.addAttribute("sucesso", "Pedido de apadrinhamento e prestação de serviço cadastrado com sucesso! Agenda criada [RN009].");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        Optional<Animal> animalOpt = adocaoService.buscarAnimalPorId(animalId);
        animalOpt.ifPresent(a -> model.addAttribute("animal", a));
        model.addAttribute("autenticado", true);
        return "apadrinhamento/animal";
    }

    @GetMapping("/apadrinhamento/financeiro")
    public String doacaoFinanceiraForm(Authentication authentication, Model model) {
        model.addAttribute("autenticado", authentication != null);
        return "apadrinhamento/financeiro";
    }

    @PostMapping("/apadrinhamento/financeiro")
    public String gerarBoleto(
            @RequestParam(required = false) String nome,
            @RequestParam(required = false) String email,
            @RequestParam Double valor,
            @RequestParam String frequencia,
            Authentication authentication,
            Model model) {
        String emailLogado = authentication != null ? authentication.getName() : null;
        DoacaoFinanceira doacao = apadrinhamentoService.gerarBoletoDoacao(nome, email, valor, frequencia, emailLogado);
        model.addAttribute("doacao", doacao);
        model.addAttribute("sucesso", "Boleto gerado com sucesso [FA-16]!");
        return "apadrinhamento/boleto";
    }
}
