package com.tcc.iterativa.service;

import com.tcc.iterativa.model.*;
import com.tcc.iterativa.repository.*;
import org.springframework.stereotype.Service;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.LocalTime;
import java.util.List;
import java.util.Optional;
import java.util.UUID;

@Service
public class ApadrinhamentoService {

    private final ApadrinhamentoRepository apadrinhamentoRepository;
    private final DoacaoFinanceiraRepository doacaoFinanceiraRepository;
    private final AgendaAnimalRepository agendaAnimalRepository;
    private final AnimalRepository animalRepository;
    private final UsuarioRepository usuarioRepository;

    public ApadrinhamentoService(
            ApadrinhamentoRepository apadrinhamentoRepository,
            DoacaoFinanceiraRepository doacaoFinanceiraRepository,
            AgendaAnimalRepository agendaAnimalRepository,
            AnimalRepository animalRepository,
            UsuarioRepository usuarioRepository) {
        this.apadrinhamentoRepository = apadrinhamentoRepository;
        this.doacaoFinanceiraRepository = doacaoFinanceiraRepository;
        this.agendaAnimalRepository = agendaAnimalRepository;
        this.animalRepository = animalRepository;
        this.usuarioRepository = usuarioRepository;
    }

    public Apadrinhamento criarApadrinhamentoServico(Long animalId, String emailAdotante, String tipoServico, LocalDate dia, LocalTime horario) throws Exception {
        Optional<Animal> animalOpt = animalRepository.findById(animalId);
        if (animalOpt.isEmpty()) {
            throw new IllegalArgumentException("Animal não encontrado.");
        }
        Optional<Usuario> usuarioOpt = usuarioRepository.findByEmail(emailAdotante);
        if (usuarioOpt.isEmpty()) {
            throw new IllegalArgumentException("Usuário não encontrado.");
        }

        Animal animal = animalOpt.get();
        Usuario adotante = usuarioOpt.get();

        Apadrinhamento ap = new Apadrinhamento(animal, adotante, tipoServico, dia, horario, "SOLICITADO", LocalDateTime.now());
        Apadrinhamento salvo = apadrinhamentoRepository.save(ap);

        // RN009: Criar agenda do animal ao receber proposta de prestação de serviço
        AgendaAnimal agenda = new AgendaAnimal(animal, dia, horario, "Prestação de Serviço: " + tipoServico);
        agendaAnimalRepository.save(agenda);

        return salvo;
    }

    public DoacaoFinanceira gerarBoletoDoacao(String nome, String email, Double valor, String frequencia, String emailLogado) {
        String nomeFinal = nome;
        String emailFinal = email;
        if (emailLogado != null && !emailLogado.isEmpty()) {
            Optional<Usuario> uOpt = usuarioRepository.findByEmail(emailLogado);
            if (uOpt.isPresent()) {
                nomeFinal = uOpt.get().getNome();
                emailFinal = uOpt.get().getEmail();
            }
        }

        String codigoBoleto = "BOLETO-" + UUID.randomUUID().toString().substring(0, 8).toUpperCase();
        DoacaoFinanceira doacao = new DoacaoFinanceira(nomeFinal, emailFinal, valor, frequencia, codigoBoleto, LocalDateTime.now());
        return doacaoFinanceiraRepository.save(doacao);
    }

    public List<AgendaAnimal> listarAgendaAnimal(Long animalId) {
        Optional<Animal> animalOpt = animalRepository.findById(animalId);
        if (animalOpt.isPresent()) {
            return agendaAnimalRepository.findByAnimal(animalOpt.get());
        }
        return List.of();
    }
}
