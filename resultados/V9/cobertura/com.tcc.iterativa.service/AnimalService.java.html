package com.tcc.iterativa.service;

import com.tcc.iterativa.model.Animal;
import com.tcc.iterativa.model.Baia;
import com.tcc.iterativa.repository.AnimalRepository;
import com.tcc.iterativa.repository.BaiaRepository;
import org.springframework.stereotype.Service;
import java.util.List;

@Service
public class AnimalService {

    private final AnimalRepository animalRepository;
    private final BaiaRepository baiaRepository;

    public AnimalService(AnimalRepository animalRepository, BaiaRepository baiaRepository) {
        this.animalRepository = animalRepository;
        this.baiaRepository = baiaRepository;
    }

    public Animal cadastrar(
            String especie, String sexo, String castrado, String vermifugado,
            String idadeAproximada, String raca, String temperamento, String porte,
            String fotos, String nomeReferencia, String historia, String patrimonioTombado,
            String emTratamento, String disponivelAdocao, String vacinasJson, Long baiaId,
            String outrasInformacoes, String autorFotografia, String contatoAutor) throws Exception {

        if (especie == null || especie.trim().isEmpty() ||
            sexo == null || sexo.trim().isEmpty() ||
            castrado == null || castrado.trim().isEmpty() ||
            vermifugado == null || vermifugado.trim().isEmpty() ||
            raca == null || raca.trim().isEmpty() ||
            nomeReferencia == null || nomeReferencia.trim().isEmpty() ||
            historia == null || historia.trim().isEmpty() ||
            patrimonioTombado == null || patrimonioTombado.trim().isEmpty() ||
            emTratamento == null || emTratamento.trim().isEmpty() ||
            disponivelAdocao == null || disponivelAdocao.trim().isEmpty() ||
            autorFotografia == null || autorFotografia.trim().isEmpty()) {
            throw new IllegalArgumentException("Preencha todos os campos obrigatórios do animal.");
        }

        // Regras de Negócio automáticas (RN003)
        if ("Sim".equals(patrimonioTombado)) {
            disponivelAdocao = "Não";
        }
        if ("Sim".equals(emTratamento)) {
            disponivelAdocao = "Não";
        }

        Baia baia = null;
        if (baiaId != null) {
            baia = baiaRepository.findById(baiaId).orElse(null);
        }

        Animal animal = new Animal();
        animal.setEspecie(especie);
        animal.setSexo(sexo);
        animal.setCastrado(castrado);
        animal.setVermifugado(vermifugado);
        animal.setIdadeAproximada(idadeAproximada);
        animal.setRaca(raca);
        animal.setTemperamento(temperamento);
        animal.setPorte(porte);
        animal.setFotos(fotos);
        animal.setNomeReferencia(nomeReferencia);
        animal.setHistoria(historia);
        animal.setPatrimonioTombado(patrimonioTombado);
        animal.setEmTratamento(emTratamento);
        animal.setDisponivelAdocao(disponivelAdocao);
        animal.setVacinasJson(vacinasJson);
        animal.setBaia(baia);
        animal.setOutrasInformacoes(outrasInformacoes);
        animal.setAutorFotografia(autorFotografia);
        animal.setContatoAutor(contatoAutor);

        return animalRepository.save(animal);
    }

    public List<Animal> listarTodos() {
        return animalRepository.findAll();
    }
}
