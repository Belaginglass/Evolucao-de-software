package com.tcc.iterativa.controller;

import com.tcc.iterativa.service.ConteudoHomeService;
import org.springframework.security.core.Authentication;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;

@Controller
public class HomeController {

    private final ConteudoHomeService conteudoHomeService;

    public HomeController(ConteudoHomeService conteudoHomeService) {
        this.conteudoHomeService = conteudoHomeService;
    }

    @GetMapping("/")
    public String index(Model model) {
        model.addAttribute("sobreNos", conteudoHomeService.obterSobreNos());
        model.addAttribute("servicos", conteudoHomeService.selecionarAleatorio(conteudoHomeService.listarServicos(), 3));
        model.addAttribute("campanhas", conteudoHomeService.selecionarAleatorio(conteudoHomeService.listarCampanhas(), 3));
        model.addAttribute("doacoes", conteudoHomeService.selecionarAleatorio(conteudoHomeService.listarDoacoes(), 3));
        return "index";
    }

    @GetMapping("/home/sobre-nos")
    public String editarSobreNosForm(Authentication authentication, Model model) {
        if (authentication == null) return "redirect:/login";
        model.addAttribute("texto", conteudoHomeService.obterSobreNos());
        return "home/editar-sobre";
    }

    @PostMapping("/home/sobre-nos")
    public String salvarSobreNos(Authentication authentication, @RequestParam String texto, Model model) {
        if (authentication == null) return "redirect:/login";
        conteudoHomeService.salvarSobreNos(texto);
        model.addAttribute("sucesso", "Texto 'sobre nós' atualizado com sucesso! [PRT-31]");
        model.addAttribute("texto", texto);
        return "home/editar-sobre";
    }

    @GetMapping("/home/servicos")
    public String gerenciarServicos(Authentication authentication, Model model) {
        if (authentication == null) return "redirect:/login";
        model.addAttribute("servicos", conteudoHomeService.listarServicos());
        return "home/editar-servicos";
    }

    @PostMapping("/home/servicos")
    public String adicionarServico(Authentication authentication, @RequestParam String titulo, @RequestParam String descricao, Model model) {
        if (authentication == null) return "redirect:/login";
        try {
            conteudoHomeService.salvarItem("SERVICO", titulo, descricao, null);
            model.addAttribute("sucesso", "Serviço adicionado com sucesso! [PRT-32]");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("servicos", conteudoHomeService.listarServicos());
        return "home/editar-servicos";
    }

    @PostMapping("/home/servicos/deletar")
    public String deletarServico(Authentication authentication, @RequestParam Long id) {
        if (authentication == null) return "redirect:/login";
        conteudoHomeService.deletarItem(id);
        return "redirect:/home/servicos";
    }

    @GetMapping("/home/campanhas")
    public String gerenciarCampanhas(Authentication authentication, Model model) {
        if (authentication == null) return "redirect:/login";
        model.addAttribute("campanhas", conteudoHomeService.listarCampanhas());
        return "home/editar-campanhas";
    }

    @PostMapping("/home/campanhas")
    public String adicionarCampanha(Authentication authentication, @RequestParam String titulo, @RequestParam String descricao, Model model) {
        if (authentication == null) return "redirect:/login";
        try {
            conteudoHomeService.salvarItem("CAMPANHA", titulo, descricao, null);
            model.addAttribute("sucesso", "Campanha adicionada com sucesso! [PRT-33]");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("campanhas", conteudoHomeService.listarCampanhas());
        return "home/editar-campanhas";
    }

    @PostMapping("/home/campanhas/deletar")
    public String deletarCampanha(Authentication authentication, @RequestParam Long id) {
        if (authentication == null) return "redirect:/login";
        conteudoHomeService.deletarItem(id);
        return "redirect:/home/campanhas";
    }

    @GetMapping("/home/doacoes")
    public String gerenciarDoacoes(Authentication authentication, Model model) {
        if (authentication == null) return "redirect:/login";
        model.addAttribute("doacoes", conteudoHomeService.listarDoacoes());
        return "home/editar-doacoes";
    }

    @PostMapping("/home/doacoes")
    public String adicionarDoacao(Authentication authentication, @RequestParam String titulo, @RequestParam String valor, Model model) {
        if (authentication == null) return "redirect:/login";
        try {
            conteudoHomeService.salvarItem("DOACAO", titulo, null, valor);
            model.addAttribute("sucesso", "Valor de doação adicionado com sucesso! [PRT-34]");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("doacoes", conteudoHomeService.listarDoacoes());
        return "home/editar-doacoes";
    }

    @PostMapping("/home/doacoes/deletar")
    public String deletarDoacao(Authentication authentication, @RequestParam Long id) {
        if (authentication == null) return "redirect:/login";
        conteudoHomeService.deletarItem(id);
        return "redirect:/home/doacoes";
    }
}
