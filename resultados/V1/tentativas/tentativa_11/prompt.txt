===== ERRO DE COMPILAÇÃO =====
O login (POST /login) autentica corretamente e redireciona (302) para /home-admin ou /home-adotante conforme o papel do usuario, mas ao seguir esse redirect, o Spring Security bloqueia o acesso (HTTP 403 Forbidden). Causa raiz: SecurityConfig define '.anyRequest().authenticated()' para as rotas /home-admin e /home-adotante, mas AuthController.fazerLogin() nunca estabelece uma autenticacao no SecurityContext do Spring Security (apenas verifica a senha no banco via UsuarioService e faz um redirect comum) -- ou seja, nenhuma sessao autenticada e criada. Corrija para que, apos um login bem-sucedido, o usuario efetivamente consiga acessar a pagina de destino (/home-admin ou /home-adotante), autenticando a sessao no Spring Security (por exemplo, usando SecurityContextHolder/SecurityContextRepository para registrar a autenticacao na sessao HTTP apos validar a senha em UsuarioService), sem reintroduzir o formLogin padrao do Spring Security (a pagina de login customizada deve continuar sendo a unica pagina de autenticacao, conforme a decisao C.1 do Apendice C).

package com.tcc.iterativa.config;

import org.springframework.stereotype.Component;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;
import java.util.Base64;

@Component
public class PasswordUtils {

    private static final int SALT_BYTES = 1; // 8 bits = 1 byte

    public String generateSalt() {
        byte[] salt = new byte[SALT_BYTES];
        new SecureRandom().nextBytes(salt);
        return Base64.getEncoder().encodeToString(salt);
    }

    public String hashPassword(String password, String saltStr) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] salt = Base64.getDecoder().decode(saltStr);
            digest.update(salt);
            byte[] hash = digest.digest(password.getBytes(StandardCharsets.UTF_8));
            return Base64.getEncoder().encodeToString(hash);
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException("Erro ao criptografar senha", e);
        }
    }

    public boolean validatePassword(String rawPassword, String saltStr, String storedHash) {
        String newHash = hashPassword(rawPassword, saltStr);
        return newHash.equals(storedHash);
    }

    public boolean validateFormat(String email, String password) {
        if (email == null || password == null) {
            return false;
        }
        // RN001: E-mail deve conter apenas letras A-Z, digitos 0-9 e exatamente um "@", sem < > " ' / \ : |
        // Senha: min 8 caracteres, sem esses mesmos caracteres.
        String emailRegex = "^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,6}$Rs*";
        // Ajuste simples para caracteres proibidos
        String forbiddenChars = "[<>\"'/:|\\\\]";
        
        if (email.matches(".*" + forbiddenChars + ".*")) return false;
        if (password.matches(".*" + forbiddenChars + ".*")) return false;
        
        long atCount = email.chars().filter(ch -> ch == '@').count();
        if (atCount != 1) return false;

        if (password.length() < 8) return false;

        return true;
    }
}

package com.tcc.iterativa.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
            .csrf(csrf -> csrf.disable())
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/", "/login", "/cadastro", "/recuperar-senha", "/atualizar-senha", "/h2-console/**").permitAll()
                .anyRequest().authenticated()
            )
            .formLogin(form -> form.disable());
        
        return http.build();
    }
}

package com.tcc.iterativa.controller;

import com.tcc.iterativa.model.Usuario;
import com.tcc.iterativa.service.UsuarioService;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;

import java.util.Optional;

@Controller
public class AuthController {

    private final UsuarioService usuarioService;

    public AuthController(UsuarioService usuarioService) {
        this.usuarioService = usuarioService;
    }

    @GetMapping({"/", "/login"})
    public String loginPage() {
        return "auth/login";
    }

    @PostMapping("/login")
    public String fazerLogin(@RequestParam String email, @RequestParam String senha, Model model) {
        Optional<Usuario> usuario = usuarioService.autenticar(email, senha);
        if (usuario.isPresent()) {
            if ("ADMIN".equals(usuario.get().getPapel())) {
                return "redirect:/home-admin";
            } else {
                return "redirect:/home-adotante";
            }
        }
        model.addAttribute("erro", "E-mail ou senha incorretos.");
        return "auth/login";
    }

    @PostMapping("/cadastro")
    public String cadastrar(@RequestParam String email, @RequestParam String senha, @RequestParam(defaultValue = "ADOTANTE") String papel, Model model) {
        try {
            usuarioService.cadastrar(email, senha, papel);
            model.addAttribute("sucesso", "Cadastro realizado com sucesso! Faça login.");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        return "auth/login";
    }

    @PostMapping("/recuperar-senha")
    public String recuperarSenha(@RequestParam String email, Model model) {
        String token = usuarioService.gerarTokenRecuperacao(email);
        if (token != null) {
            model.addAttribute(
                "sucesso",
                "Link de recuperação enviado. Simulação de link: /atualizar-senha?token=" + token
            );
        } else {
            model.addAttribute("erro", "E-mail não encontrado.");
        }
        return "auth/login";
    }

    @GetMapping("/atualizar-senha")
    public String atualizarSenhaPage(@RequestParam String token, Model model) {
        Optional<Usuario> u = usuarioService.buscarPorToken(token);
        if (u.isPresent()) {
            model.addAttribute("token", token);
            return "auth/nova-senha";
        }
        model.addAttribute("erro", "Token inválido ou expirado.");
        return "auth/login";
    }

    @PostMapping("/atualizar-senha")
    public String atualizarSenha(@RequestParam String token, @RequestParam String novaSenha, Model model) {
        boolean sucesso = usuarioService.alterarSenhaComToken(token, novaSenha);
        if (sucesso) {
            model.addAttribute("sucesso", "Senha alterada com sucesso! Faça login.");
            return "auth/login";
        }
        model.addAttribute("token", token);
        model.addAttribute("erro", "Falha ao alterar senha. Verifique o formato ou validade.");
        return "auth/nova-senha";
    }

    @GetMapping("/home-admin")
    public String homeAdmin() {
        return "home/admin";
    }

    @GetMapping("/home-adotante")
    public String homeAdotante() {
        return "home/adotante";
    }
}

package com.tcc.iterativa.service;

import com.tcc.iterativa.config.PasswordUtils;
import com.tcc.iterativa.model.Usuario;
import com.tcc.iterativa.repository.UsuarioRepository;
import org.springframework.stereotype.Service;

import java.time.LocalDateTime;
import java.util.Optional;
import java.util.UUID;

@Service
public class UsuarioService {

    private final UsuarioRepository usuarioRepository;
    private final PasswordUtils passwordUtils;

    public UsuarioService(UsuarioRepository usuarioRepository, PasswordUtils passwordUtils) {
        this.usuarioRepository = usuarioRepository;
        this.passwordUtils = passwordUtils;
    }

    public Usuario cadastrar(String email, String senha, String papel) throws Exception {
        if (!passwordUtils.validateFormat(email, senha)) {
            throw new IllegalArgumentException("E-mail ou senha inválidos segundo as regras de formato.");
        }
        if (usuarioRepository.findByEmail(email).isPresent()) {
            throw new IllegalArgumentException("E-mail já cadastrado.");
        }

        String salt = passwordUtils.generateSalt();
        String hash = passwordUtils.hashPassword(senha, salt);

        Usuario usuario = new Usuario(email, hash, salt, papel);
        return usuarioRepository.save(usuario);
    }

    public Optional<Usuario> autenticar(String email, String senha) {
        Optional<Usuario> opt = usuarioRepository.findByEmail(email);
        if (opt.isPresent()) {
            Usuario u = opt.get();
            if (passwordUtils.validatePassword(senha, u.getSalt(), u.getSenha())) {
                return Optional.of(u);
            }
        }
        return Optional.empty();
    }

    public String gerarTokenRecuperacao(String email) {
        Optional<Usuario> opt = usuarioRepository.findByEmail(email);
        if (opt.isPresent()) {
            Usuario u = opt.get();
            String token = UUID.randomUUID().toString().replace("-", ""); // 32 caracteres
            u.getTokenRecuperacao();
            u.setTokenRecuperacao(token);
            u.setValidadeToken(LocalDateTime.now().plusHours(24));
            usuarioRepository.save(u);
            return token;
        }
        return null;
    }

    public boolean alterarSenhaComToken(String token, String novaSenha) {
        Optional<Usuario> opt = usuarioRepository.findByTokenRecuperacao(token);
        if (opt.isPresent()) {
            Usuario u = opt.get();
            if (u.getValidadeToken() != null && u.getValidadeToken().isAfter(LocalDateTime.now())) {
                if (!passwordUtils.validateFormat(u.getEmail(), novaSenha)) {
                    return false;
                }
                String salt = passwordUtils.generateSalt();
                u.setSenha(passwordUtils.hashPassword(novaSenha, salt));
                u.setSalt(salt);
                u.setTokenRecuperacao(null);
                u.setValidadeToken(null);
                usuarioRepository.save(u);
                return true;
            }
        }
        return false;
    }
    
    public Optional<Usuario> buscarPorToken(String token) {
        Optional<Usuario> opt = usuarioRepository.findByTokenRecuperacao(token);
        if (opt.isPresent() && opt.get().getValidadeToken() != null && opt.get().getValidadeToken().isAfter(LocalDateTime.now())) {
            return opt;
        }
        return Optional.empty();
    }
}
