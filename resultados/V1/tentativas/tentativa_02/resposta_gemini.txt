+package com.tcc.iterativa.config;
+
+import org.springframework.stereotype.Component;
+
+import java.nio.charset.StandardCharsets;
+import java.security.MessageDigest;
+import java.security.NoSuchAlgorithmException;
+import java.security.SecureRandom;
+import java.util.Base64;
+
+@Component
+public class PasswordConfig {
+
+    private static final int SALT_BYTES = 1; // 8 bits = 1 byte
+
+    public String generateSalt() {
+        SecureRandom random = new SecureRandom();
+        byte[] salt = new byte[SALT_BYTES];
+        random.nextBytes(salt);
+        return Base64.getEncoder().encodeToString(salt);
+    }
+
+    public String hashPassword(String password, String saltStr) {
+        try {
+            MessageDigest digest = MessageDigest.getInstance("SHA-256");
+            byte[] salt = Base64.getDecoder().decode(saltStr);
+            digest.update(salt);
+            byte[] hash = digest.digest(password.getBytes(StandardCharsets.UTF_8));
+            return Base64.getEncoder().encodeToString(hash);
+        } catch (NoSuchAlgorithmException e) {
+            throw new RuntimeException("Erro ao criptografar senha com SHA-256", e);
+        }
+    }
+
+    public String hashPasswordWithNewSalt(String password, String[] outSaltContainer) {
+        String salt = generateSalt();
+        outSaltContainer[0] = salt;
+        return hashPassword(password, salt);
+    }
+
+    public boolean verifyPassword(String rawPassword, String storedHash, String storedSalt) {
+        if (storedHash == null || storedSalt == null) {
+            return false;
+        }
+        String computedHash = hashPassword(rawPassword, storedSalt);
+        return computedHash.equals(storedHash);
+    }
+}

+package com.tcc.iterativa.service;
+
+import com.tcc.iterativa.config.PasswordConfig;
+import com.tcc.iterativa.model.Usuario;
+import com.tcc.iterativa.repository.UsuarioRepository;
+import org.springframework.beans.factory.annotation.Autowired;
+import org.springframework.stereotype.Service;
+
+import java.time.LocalDate;
+import java.util.Optional;
+import java.util.UUID;
+import java.util.regex.Pattern;
+
+@Service
+public class AuthService {
+
+    @Autowired
+    private UsuarioRepository usuarioRepository;
+
+    @Autowired
+    private PasswordConfig passwordConfig;
+
+    private static final Pattern EMAIL_PATTERN = Pattern.compile("^[A-Za-z0-9]+@[A-Za-z0-9]+(\\.[A-Za-z0-9]+)*$");
+
+    public void validarEmailESenha(String email, String senha) {
+        if (email == null || !EMAIL_PATTERN.matcher(email).matches() || contemCaracteresProibidos(email)) {
+            throw new IllegalArgumentException("E-mail inválido conforme regras de negócio.");
+        }
+        if (senha == null || senha.length() < 8 || contemCaracteresProibidos(senha)) {
+            throw new IllegalArgumentException("Senha inválida: deve ter no mínimo 8 caracteres e não conter caracteres proibidos (<, >, \", ', /, \\, :, |).");
+        }
+    }
+
+    private boolean contemCaracteresProibidos(String valor) {
+        String proibidos = "<>\"'/\\:|";
+        for (char c : proibidos.toCharArray()) {
+            if (valor.indexOf(c) >= 0) {
+                return true;
+            }
+        }
+        return false;
+    }
+
+    public Usuario cadastrar(String email, String senha, boolean administrativo) {
+        validarEmailESenha(email, senha);
+        if (usuarioRepository.findByEmail(email).isPresent()) {
+            throw new IllegalArgumentException("E-mail já cadastrado no sistema.");
+        }
+        String[] saltContainer = new String[1];
+        String hash = passwordConfig.hashPasswordWithNewSalt(senha, saltContainer);
+        Usuario usuario = new Usuario(email, hash, saltContainer[0], administrativo);
+        return usuarioRepository.save(usuario);
+    }
+
+    public Usuario autenticar(String email, String senha) {
+        validarEmailESenha(email, senha);
+        Usuario usuario = usuarioRepository.findByEmail(email)
+                .orElseThrow(() -> new IllegalArgumentException("E-mail ou senha incorretos."));
+
+        boolean valido = passwordConfig.verifyPassword(senha, usuario.getSenhaHash(), usuario.getSalt());
+        if (!valido) {
+            throw new IllegalArgumentException("E-mail ou senha incorretos.");
+        }
+        return usuario;
+    }
+
+    public void solicitarRecuperacaoSenha(String email) {
+        Optional<Usuario> opt = usuarioRepository.findByEmail(email);
+        if (opt.isPresent()) {
+            Usuario usuario = opt.get();
+            String token = UUID.randomUUID().toString(); // 36 caracteres UUID, atendendo RN005
+            usuario.setTokenRecuperacao(token);
+            usuario.setDataToken(LocalDate.now());
+            usuarioRepository.save(usuario);
+            // Envio simulado de e-mail (gravado no log/banco)
+        }
+    }
+
+    public boolean validarToken(String token) {
+        Optional<Usuario> opt = usuarioRepository.findByTokenRecuperacao(token);
+        if (opt.isEmpty()) {
+            return false;
+        }
+        Usuario usuario = opt.get();
+        LocalDate dataToken = usuario.getDataToken();
+        if (dataToken == null || dataToken.isBefore(LocalDate.now().minusDays(1))) {
+            return false; // Expirado (validade de 24h)
+        }
+        return true;
+    }
+
+    public void alterarSenhaComToken(String token, String novaSenha) {
+        Usuario usuario = usuarioRepository.findByTokenRecuperacao(token)
+                .orElseThrow(() -> new IllegalArgumentException("Token inválido."));
+        validarEmailESenha(usuario.getEmail(), novaSenha);
+        String[] saltContainer = new String[1];
+        String hash = passwordConfig.hashPasswordWithNewSalt(novaSenha, saltContainer);
+        usuario.setSenhaHash(hash);
+        usuario.setSalt(saltContainer[0]);
+        usuario.setTokenRecuperacao(null);
+        usuario.setDataToken(null);
+        usuarioRepository.save(usuario);
+    }
+}
