package com.tcc.iterativa.controller;

import com.tcc.iterativa.model.Animal;
import com.tcc.iterativa.service.AnimalService;
import com.tcc.iterativa.service.BaiaService;
import org.springframework.security.core.Authentication;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.multipart.MultipartFile;

import java.util.Base64;

@Controller
public class AnimalController {

    private final AnimalService animalService;
    private final BaiaService baiaService;

    public AnimalController(AnimalService animalService, BaiaService baiaService) {
        this.animalService = animalService;
        this.baiaService = baiaService;
    }

    @GetMapping("/animais/novo")
    public String cadastrarAnimalForm(Authentication authentication, Model model) {
        if (authentication == null) return "redirect:/login";
        model.addAttribute("baias", baiaService.listarTodas());
        return "animal/cadastro";
    }

    @PostMapping("/animais/novo")
    public String cadastrarAnimal(
            Authentication authentication,
            @RequestParam String especie,
            @RequestParam String sexo,
            @RequestParam String castrado,
            @RequestParam String vermifugado,
            @RequestParam(required = false) String idadeAproximada,
            @RequestParam String raca,
            @RequestParam(required = false) String[] temperamento,
            @RequestParam(required = false) String porte,
            @RequestParam(required = false) MultipartFile[] fotosFiles,
            @RequestParam String nomeReferencia,
            @RequestParam String historia,
            @RequestParam String patrimonioTombado,
            @RequestParam String emTratamento,
            @RequestParam String disponivelAdocao,
            @RequestParam(required = false) String vacinasJson,
            @RequestParam(required = false) Long baiaId,
            @RequestParam(required = false) String outrasInformacoes,
            @RequestParam String autorFotografia,
            @RequestParam(required = false) String contatoAutor,
            Model model) {

        if (authentication == null) return "redirect:/login";

        try {
            String tempStr = (temperamento != null) ? String.join(", ", temperamento) : null;
            
            String fotosStr = null;
            if (fotosFiles != null && fotosFiles.length > 0) {
                StringBuilder sb = new StringBuilder();
                for (MultipartFile f : fotosFiles) {
                    if (!f.isEmpty()) {
                        if (sb.length() > 0) sb.append(";");
                        sb.append(Base64.getEncoder().encodeToString(f.getBytes()));
                    }
                }
                fotosStr = sb.length() > 0 ? sb.toString() : null;
            }

            animalService.cadastrar(
                especie, sexo, castrado, vermifugado, idadeAproximada, raca,
                tempStr, porte, fotosStr, nomeReferencia, historia,
                patrimonioTombado, emTratamento, disponivelAdocao,
                vacinasJson, baiaId, outrasInformacoes, autorFotografia, contatoAutor
            );

            model.addAttribute("sucesso", "Animal cadastrado com sucesso! [PRT-12]");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("baias", baiaService.listarTodas());
        return "animal/cadastro";
    }
}
