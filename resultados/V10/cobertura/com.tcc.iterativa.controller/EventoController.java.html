package com.tcc.iterativa.controller;

import com.tcc.iterativa.model.Evento;
import com.tcc.iterativa.service.EventoDoacaoRelatorioService;
import org.springframework.security.core.Authentication;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.multipart.MultipartFile;

import java.time.LocalDate;
import java.util.Optional;

@Controller
public class EventoController {

    private final EventoDoacaoRelatorioService service;

    public EventoController(EventoDoacaoRelatorioService service) {
        this.service = service;
    }

    @GetMapping("/eventos")
    public String listarEventosPublico(Model model) {
        model.addAttribute("eventos", service.listarEventosPublicos());
        return "evento/publico";
    }

    @GetMapping("/eventos/{id}")
    public String detalheEvento(@PathVariable Long id, Model model) {
        Optional<Evento> evento = service.buscarEventoPorId(id);
        if (evento.isEmpty()) return "redirect:/eventos";
        model.addAttribute("evento", evento.get());
        return "evento/detalhe";
    }

    @GetMapping("/eventos/gerenciar")
    public String gerenciarEventos(Authentication authentication, Model model) {
        if (authentication == null) return "redirect:/login";
        model.addAttribute("eventos", service.listarEventosPublicos());
        return "evento/gerenciar";
    }

    @PostMapping("/eventos/salvar")
    public String salvarEvento(
            Authentication authentication,
            @RequestParam(required = false) Long id,
            @RequestParam String nome,
            @RequestParam LocalDate data,
            @RequestParam String descricao,
            @RequestParam(required = false) MultipartFile[] fotosFiles,
            Model model) {
        if (authentication == null) return "redirect:/login";
        try {
            if (id == null) {
                service.salvarEvento(nome, data, descricao, fotosFiles);
                model.addAttribute("sucesso", "Evento criado com sucesso! [PRT-45/46]");
            } else {
                service.atualizarEvento(id, nome, data, descricao, fotosFiles);
                model.addAttribute("sucesso", "Evento atualizado com sucesso!");
            }
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("eventos", service.listarEventosPublicos());
        return "evento/gerenciar";
    }

    @PostMapping("/eventos/deletar")
    public String deletarEvento(Authentication authentication, @RequestParam Long id) {
        if (authentication == null) return "redirect:/login";
        service.deletarEvento(id);
        return "redirect:/eventos/gerenciar";
    }
}
