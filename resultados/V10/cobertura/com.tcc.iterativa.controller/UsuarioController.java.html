package com.tcc.iterativa.controller;

import com.tcc.iterativa.model.Usuario;
import com.tcc.iterativa.repository.UsuarioRepository;
import com.tcc.iterativa.service.UsuarioService;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.multipart.MultipartFile;

import java.time.LocalDate;
import java.util.List;
import java.util.Optional;

@Controller
public class UsuarioController {

    private final UsuarioService usuarioService;
    private final UsuarioRepository usuarioRepository;

    public UsuarioController(UsuarioService usuarioService, UsuarioRepository usuarioRepository) {
        this.usuarioService = usuarioService;
        this.usuarioRepository = usuarioRepository;
    }

    @GetMapping("/meus-dados")
    public String exibirMeusDados(Authentication authentication, Model model) {
        if (authentication == null) return "redirect:/login";
        String email = authentication.getName();
        Optional<Usuario> opt = usuarioRepository.findByEmail(email);
        if (opt.isPresent()) {
            model.addAttribute("usuario", opt.get());
            return "auth/meus-dados";
        }
        return "redirect:/login";
    }

    @PostMapping("/meus-dados")
    public String atualizarMeusDados(
            Authentication authentication,
            @RequestParam String nome,
            @RequestParam String cpf,
            @RequestParam LocalDate dataNascimento,
            @RequestParam String telefones,
            @RequestParam String rua,
            @RequestParam String numero,
            @RequestParam(required = false) String complemento,
            @RequestParam String bairro,
            @RequestParam String cidade,
            @RequestParam String estado,
            @RequestParam String cep,
            @RequestParam(required = false) MultipartFile fotoFile,
            @RequestParam String voluntario,
            @RequestParam(required = false) String novaSenha,
            Model model) {
        if (authentication == null) return "redirect:/login";
        String email = authentication.getName();
        try {
            byte[] fotoBytes = (fotoFile != null && !fotoFile.isEmpty()) ? fotoFile.getBytes() : null;
            String token = usuarioService.atualizarDadosPessoais(email, nome, cpf, dataNascimento, telefones, rua, numero, complemento, bairro, cidade, estado, cep, fotoBytes, voluntario, novaSenha);
            model.addAttribute("sucesso", "Dados atualizados com sucesso! E-mail de confirmação simulado com token: /atualizar-senha?token=" + token);
            Optional<Usuario> opt = usuarioRepository.findByEmail(email);
            opt.ifPresent(usuario -> model.addAttribute("usuario", usuario));
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
            Optional<Usuario> opt = usuarioRepository.findByEmail(email);
            opt.ifPresent(usuario -> model.addAttribute("usuario", usuario));
        }
        return "auth/meus-dados";
    }

    @GetMapping("/usuarios/gerenciar")
    public String gerenciarAtores(
            Authentication authentication,
            @RequestParam(required = false) String busca,
            @RequestParam(required = false) String cargoFiltro,
            Model model) {
        if (authentication == null) return "redirect:/login";
        
        List<Usuario> usuarios = usuarioService.listarUsuariosAdministrativos(busca, cargoFiltro);
        model.addAttribute("usuarios", usuarios);
        model.addAttribute("busca", busca);
        model.addAttribute("cargoFiltro", cargoFiltro);
        return "usuario/gerenciar";
    }

    @PostMapping("/usuarios/privilegios")
    public String alterarPrivilegios(
            Authentication authentication,
            @RequestParam Long usuarioId,
            @RequestParam String papel,
            @RequestParam String statusAdmin,
            Model model) {
        if (authentication == null) return "redirect:/login";
        
        if (authentication.getAuthorities() == null || !authentication.getAuthorities().contains(new SimpleGrantedAuthority("ROLE_DIRETOR_GERAL"))) {
            model.addAttribute("erro", "Acesso negado: apenas Diretor Geral pode alterar privilégios.");
            List<Usuario> usuarios = usuarioService.listarUsuariosAdministrativos(null, null);
            model.addAttribute("usuarios", usuarios);
            return "usuario/gerenciar";
        }
        
        try {
            usuarioService.alterarPrivilegios(usuarioId, papel, statusAdmin);
            model.addAttribute("sucesso", "Privilégios atualizados com sucesso! [PRT-10]");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        
        List<Usuario> usuarios = usuarioService.listarUsuariosAdministrativos(null, null);
        model.addAttribute("usuarios", usuarios);
        return "usuario/gerenciar";
    }

    @PostMapping("/usuarios/excluir")
    public String excluirAcessoAdministrativo(
            Authentication authentication,
            @RequestParam Long usuarioId,
            Model model) {
        if (authentication == null) return "redirect:/login";
        
        if (authentication.getAuthorities() == null || !authentication.getAuthorities().contains(new SimpleGrantedAuthority("ROLE_DIRETOR_GERAL"))) {
            model.addAttribute("erro", "Acesso negado: apenas Diretor Geral pode excluir acesso administrativo.");
            List<Usuario> usuarios = usuarioService.listarUsuariosAdministrativos(null, null);
            model.addAttribute("usuarios", usuarios);
            return "usuario/gerenciar";
        }
        
        try {
            usuarioService.excluirAcessoAdministrativo(usuarioId);
            model.addAttribute("sucesso", "Acesso administrativo excluído com sucesso!");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        List<Usuario> usuarios = usuarioService.listarUsuariosAdministrativos(null, null);
        model.addAttribute("usuarios", usuarios);
        return "usuario/gerenciar";
    }
}
