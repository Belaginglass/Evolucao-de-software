package com.tcc.iterativa.controller;

import com.tcc.iterativa.model.PedidoDoacao;
import com.tcc.iterativa.service.EventoDoacaoRelatorioService;
import org.springframework.security.core.Authentication;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.multipart.MultipartFile;

import java.time.LocalDate;
import java.util.Optional;

@Controller
public class PedidoDoacaoController {

    private final EventoDoacaoRelatorioService service;

    public PedidoDoacaoController(EventoDoacaoRelatorioService service) {
        this.service = service;
    }

    @GetMapping("/doacoes/pedidos")
    public String listarPedidosPublico(Model model) {
        model.addAttribute("pedidos", service.listarPedidosDoacaoPublicos());
        return "doacao/pedidos-publico";
    }

    @GetMapping("/doacoes/pedidos/{id}")
    public String detalhePedido(@PathVariable Long id, Model model) {
        Optional<PedidoDoacao> pedido = service.buscarPedidoDoacaoPorId(id);
        if (pedido.isEmpty()) return "redirect:/doacoes/pedidos";
        model.addAttribute("pedido", pedido.get());
        return "doacao/detalhe-pedido";
    }

    @GetMapping("/doacoes/gerenciar")
    public String gerenciarPedidos(Authentication authentication, Model model) {
        if (authentication == null) return "redirect:/login";
        model.addAttribute("pedidos", service.listarPedidosDoacaoPublicos());
        return "doacao/gerenciar";
    }

    @PostMapping("/doacoes/gerenciar/salvar")
    public String salvarPedido(
            Authentication authentication,
            @RequestParam(required = false) Long id,
            @RequestParam String titulo,
            @RequestParam Double valorSugerido,
            @RequestParam String descricao,
            @RequestParam LocalDate dataValidade,
            @RequestParam(required = false) MultipartFile[] fotosFiles,
            Model model) {
        if (authentication == null) return "redirect:/login";
        try {
            if (id == null) {
                service.salvarPedidoDoacao(titulo, valorSugerido, descricao, dataValidade, fotosFiles);
                model.addAttribute("sucesso", "Pedido de doação criado com sucesso! [PRT-48/49]");
            } else {
                service.atualizarPedidoDoacao(id, titulo, valorSugerido, descricao, dataValidade, fotosFiles);
                model.addAttribute("sucesso", "Pedido de doação atualizado com sucesso!");
            }
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("pedidos", service.listarPedidosDoacaoPublicos());
        return "doacao/gerenciar";
    }

    @PostMapping("/doacoes/gerenciar/deletar")
    public String deletarPedido(Authentication authentication, @RequestParam Long id) {
        if (authentication == null) return "redirect:/login";
        service.deletarPedidoDoacao(id);
        return "redirect:/doacoes/gerenciar";
    }
}
