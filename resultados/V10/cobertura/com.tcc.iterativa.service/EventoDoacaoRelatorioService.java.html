package com.tcc.iterativa.service;

import com.tcc.iterativa.model.Evento;
import com.tcc.iterativa.model.PedidoDoacao;
import com.tcc.iterativa.repository.EventoRepository;
import com.tcc.iterativa.repository.PedidoDoacaoRepository;
import com.tcc.iterativa.repository.AnimalRepository;
import com.tcc.iterativa.repository.AdocaoRepository;
import com.tcc.iterativa.repository.UsuarioRepository;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import java.time.LocalDate;
import java.util.Base64;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.stream.Collectors;

@Service
public class EventoDoacaoRelatorioService {

    private final EventoRepository eventoRepository;
    private final PedidoDoacaoRepository pedidoDoacaoRepository;
    private final AnimalRepository animalRepository;
    private final AdocaoRepository adocaoRepository;
    private final UsuarioRepository usuarioRepository;

    public EventoDoacaoRelatorioService(
            EventoRepository eventoRepository,
            PedidoDoacaoRepository pedidoDoacaoRepository,
            AnimalRepository animalRepository,
            AdocaoRepository adocaoRepository,
            UsuarioRepository usuarioRepository) {
        this.eventoRepository = eventoRepository;
        this.pedidoDoacaoRepository = pedidoDoacaoRepository;
        this.animalRepository = animalRepository;
        this.adocaoRepository = adocaoRepository;
        this.usuarioRepository = usuarioRepository;
    }

    public List<Evento> listarEventosPublicos() {
        return eventoRepository.findAllByOrderByDataAsc();
    }

    public Optional<Evento> buscarEventoPorId(Long id) {
        return eventoRepository.findById(id);
    }

    public Evento salvarEvento(String nome, LocalDate data, String descricao, MultipartFile[] fotosFiles) throws Exception {
        String fotosStr = processarFotos(fotosFiles);
        Evento evento = new Evento(nome, data, descricao, fotosStr);
        return eventoRepository.save(evento);
    }

    public Evento atualizarEvento(Long id, String nome, LocalDate data, String descricao, MultipartFile[] fotosFiles) throws Exception {
        Optional<Evento> opt = eventoRepository.findById(id);
        if (opt.isEmpty()) throw new IllegalArgumentException("Evento não encontrado.");
        Evento evento = opt.get();
        evento.setNome(nome);
        evento.setData(data);
        evento.setDescricao(descricao);
        String fotosStr = processarFotos(fotosFiles);
        if (fotosStr != null) {
            evento.setFotos(fotosStr);
        }
        return eventoRepository.save(evento);
    }

    public void deletarEvento(Long id) {
        eventoRepository.deleteById(id);
    }

    public List<PedidoDoacao> listarPedidosDoacaoPublicos() {
        return pedidoDoacaoRepository.findAllByOrderByDataValidadeAsc();
    }

    public Optional<PedidoDoacao> buscarPedidoDoacaoPorId(Long id) {
        return pedidoDoacaoRepository.findById(id);
    }

    public PedidoDoacao salvarPedidoDoacao(String titulo, Double valorSugerido, String descricao, LocalDate dataValidade, MultipartFile[] fotosFiles) throws Exception {
        String fotosStr = processarFotos(fotosFiles);
        PedidoDoacao pedido = new PedidoDoacao(titulo, valorSugerido, descricao, dataValidade, fotosStr);
        return pedidoDoacaoRepository.save(pedido);
    }

    public PedidoDoacao atualizarPedidoDoacao(Long id, String titulo, Double valorSugerido, String descricao, LocalDate dataValidade, MultipartFile[] fotosFiles) throws Exception {
        Optional<PedidoDoacao> opt = pedidoDoacaoRepository.findById(id);
        if (opt.isEmpty()) throw new IllegalArgumentException("Pedido de doação não encontrado.");
        PedidoDoacao pedido = opt.get();
        pedido.setTitulo(titulo);
        pedido.setValorSugerido(valorSugerido);
        pedido.setDescricao(descricao);
        pedido.setDataValidade(dataValidade);
        String fotosStr = processarFotos(fotosFiles);
        if (fotosStr != null) {
            pedido.setFotos(fotosStr);
        }
        return pedidoDoacaoRepository.save(pedido);
    }

    public void deletarPedidoDoacao(Long id) {
        pedidoDoacaoRepository.deleteById(id);
    }

    public Map<String, Object> gerarRelatoriosConsolidados() {
        Map<String, Object> relatorios = new HashMap<>();
        relatorios.put("totalAnimais", animalRepository.count());
        relatorios.put("totalAdocoes", adocaoRepository.count());
        relatorios.put("totalUsuarios", usuarioRepository.count());
        relatorios.put("totalEventos", eventoRepository.count());
        relatorios.put("totalPedidosDoacao", pedidoDoacaoRepository.count());
        
        Map<String, Long> adocoesPorEstado = adocaoRepository.findAll().stream()
            .collect(Collectors.groupingBy(a -> a.getEstado() != null ? a.getEstado() : "DESCONHECIDO", Collectors.counting()));
        relatorios.put("adocoesPorEstado", adocoesPorEstado);
        
        return relatorios;
    }

    private String processarFotos(MultipartFile[] fotosFiles) throws Exception {
        if (fotosFiles == null || fotosFiles.length == 0) return null;
        StringBuilder sb = new StringBuilder();
        for (MultipartFile f : fotosFiles) {
            if (f != null && !f.isEmpty()) {
                if (sb.length() > 0) sb.append(";");
                sb.append(Base64.getEncoder().encodeToString(f.getBytes()));
            }
        }
        return sb.length() > 0 ? sb.toString() : null;
    }
}
