package com.tcc.iterativa.controller;

import com.tcc.iterativa.model.Usuario;
import com.tcc.iterativa.service.UsuarioService;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContext;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.web.context.HttpSessionSecurityContextRepository;
import org.springframework.security.web.context.SecurityContextRepository;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

import java.time.LocalDate;
import java.util.Collections;
import java.util.Map;
import java.util.Optional;

@Controller
public class AuthController {

    private final UsuarioService usuarioService;
    private final SecurityContextRepository securityContextRepository = new HttpSessionSecurityContextRepository();

    public AuthController(UsuarioService usuarioService) {
        this.usuarioService = usuarioService;
    }

    @GetMapping("/login")
    public String loginPage() {
        return "auth/login";
    }

    @GetMapping("/cadastro")
    public String cadastroPage() {
        return "auth/cadastro";
    }

    @PostMapping("/login")
    public String fazerLogin(@RequestParam String email, @RequestParam String senha, Model model, HttpServletRequest request, HttpServletResponse response) {
        Optional<Usuario> usuarioOpt = usuarioService.autenticar(email, senha);
        if (usuarioOpt.isPresent()) {
            Usuario usuario = usuarioOpt.get();
            
            if ("DESABILITADO".equals(usuario.getStatusAdmin())) {
                model.addAttribute("erro", "Seu acesso administrativo foi desabilitado.");
                return "auth/login";
            }
            
            UsernamePasswordAuthenticationToken authentication = new UsernamePasswordAuthenticationToken(
                usuario.getEmail(),
                null,
                Collections.singletonList(new SimpleGrantedAuthority("ROLE_" + usuario.getPapel()))
            );
            
            SecurityContext context = SecurityContextHolder.createEmptyContext();
            context.setAuthentication(authentication);
            SecurityContextHolder.setContext(context);
            securityContextRepository.saveContext(context, request, response);

            if ("DIRETOR_GERAL".equals(usuario.getPapel()) || "GESTOR_ADOCAO".equals(usuario.getPapel()) || "ADMIN".equals(usuario.getPapel())) {
                return "redirect:/home-admin";
            } else {
                return "redirect:/home-adotante";
            }
        }
        model.addAttribute("erro", "E-mail ou senha incorretos.");
        return "auth/login";
    }

    @PostMapping("/cadastro")
    public String cadastrar(
            @RequestParam String email,
            @RequestParam String senha,
            @RequestParam(defaultValue = "ADOTANTE") String papel,
            @RequestParam String nome,
            @RequestParam String cpf,
            @RequestParam LocalDate dataNascimento,
            @RequestParam String telefones,
            @RequestParam String rua,
            @RequestParam String numero,
            @RequestParam(required = false) String complemento,
            @RequestParam String bairro,
            @RequestParam String cidade,
            @RequestParam String estado,
            @RequestParam String cep,
            @RequestParam("fotoFile") MultipartFile fotoFile,
            @RequestParam String voluntario,
            Model model,
            HttpServletRequest request,
            HttpServletResponse response) {
        try {
            byte[] fotoBytes = fotoFile != null ? fotoFile.getBytes() : null;
            Usuario novoUsuario = usuarioService.cadastrar(email, senha, papel, nome, cpf, dataNascimento, telefones, rua, numero, complemento, bairro, cidade, estado, cep, fotoBytes, voluntario);
            
            UsernamePasswordAuthenticationToken authentication = new UsernamePasswordAuthenticationToken(
                novoUsuario.getEmail(),
                null,
                Collections.singletonList(new SimpleGrantedAuthority("ROLE_" + novoUsuario.getPapel()))
            );
            SecurityContext context = SecurityContextHolder.createEmptyContext();
            context.setAuthentication(authentication);
            SecurityContextHolder.setContext(context);
            securityContextRepository.saveContext(context, request, response);

            if ("DIRETOR_GERAL".equals(novoUsuario.getPapel()) || "GESTOR_ADOCAO".equals(novoUsuario.getPapel()) || "ADMIN".equals(novoUsuario.getPapel())) {
                return "redirect:/home-admin";
            } else {
                return "redirect:/home-adotante";
            }
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
            return "auth/cadastro";
        }
    }

    @GetMapping("/buscar-cep")
    @ResponseBody
    public Map<String, String> buscarCep(@RequestParam String cep) {
        return usuarioService.consultarCep(cep);
    }

    @PostMapping("/recuperar-senha")
    public String recuperarSenha(@RequestParam String email, Model model) {
        String token = usuarioService.gerarTokenRecuperacao(email);
        if (token != null) {
            model.addAttribute(
                "sucesso",
                "Link de recuperação enviado. Simulação de link: /atualizar-senha?token=" + token
            );
        } else {
            model.addAttribute("erro", "E-mail não encontrado.");
        }
        return "auth/login";
    }

    @GetMapping("/atualizar-senha")
    public String atualizarSenhaPage(@RequestParam String token, Model model) {
        Optional<Usuario> u = usuarioService.buscarPorToken(token);
        if (u.isPresent()) {
            model.addAttribute("token", token);
            return "auth/nova-senha";
        }
        model.addAttribute("erro", "Token inválido ou expirado.");
        return "auth/login";
    }

    @PostMapping("/atualizar-senha")
    public String atualizarSenha(@RequestParam String token, @RequestParam String novaSenha, Model model) {
        boolean sucesso = usuarioService.alterarSenhaComToken(token, novaSenha);
        if (sucesso) {
            model.addAttribute("sucesso", "Senha alterada com sucesso! Faça login.");
            return "auth/login";
        }
        model.addAttribute("token", token);
        model.addAttribute("erro", "Falha ao alterar senha. Verifique o formato ou validade.");
        return "auth/nova-senha";
    }

    @GetMapping("/home-admin")
    public String homeAdmin() {
        return "home/admin";
    }

    @GetMapping("/home-adotante")
    public String homeAdotante() {
        return "home/adotante";
    }
}
