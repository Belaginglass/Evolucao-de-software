package com.tcc.iterativa.controller;

import com.tcc.iterativa.model.Adocao;
import com.tcc.iterativa.model.Animal;
import com.tcc.iterativa.service.AdocaoService;
import org.springframework.security.core.Authentication;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.multipart.MultipartFile;

import java.time.LocalDate;
import java.util.Optional;

@Controller
public class AdocaoController {

    private final AdocaoService adocaoService;

    public AdocaoController(AdocaoService adocaoService) {
        this.adocaoService = adocaoService;
    }

    @GetMapping("/adote-aqui")
    public String listarAnimaisAdocao(Model model, Authentication authentication) {
        model.addAttribute("animais", adocaoService.listarAnimaisDisponiveisParaAdocao());
        model.addAttribute("autenticado", authentication != null);
        return "adocao/lista-animais";
    }

    @GetMapping("/animais/perfil/{id}")
    public String verPerfilAnimal(@PathVariable Long id, Model model, Authentication authentication) {
        Optional<Animal> animal = adocaoService.buscarAnimalPorId(id);
        if (animal.isEmpty()) {
            return "redirect:/adote-aqui";
        }
        model.addAttribute("animal", animal.get());
        model.addAttribute("autenticado", authentication != null);
        return "adocao/perfil-animal";
    }

    @GetMapping("/adocoes/termo/{animalId}")
    public String termoAdocaoForm(@PathVariable Long animalId, Authentication authentication, Model model) {
        if (authentication == null) {
            return "redirect:/login?redirect=/adocoes/termo/" + animalId;
        }
        Optional<Animal> animal = adocaoService.buscarAnimalPorId(animalId);
        if (animal.isEmpty()) {
            return "redirect:/adote-aqui";
        }
        model.addAttribute("animal", animal.get());
        return "adocao/termo";
    }

    @PostMapping("/adocoes/solicitar")
    public String solicitarAdocao(
            @RequestParam Long animalId,
            @RequestParam(required = false) String termoAceito,
            Authentication authentication,
            Model model) {
        if (authentication == null) {
            return "redirect:/login";
        }
        try {
            boolean aceito = "on".equals(termoAceito) || "true".equals(termoAceito);
            adocaoService.criarPedidoAdocao(animalId, authentication.getName(), aceito);
            return "redirect:/adocoes/minhas";
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
            Optional<Animal> animal = adocaoService.buscarAnimalPorId(animalId);
            animal.ifPresent(a -> model.addAttribute("animal", a));
            return "adocao/termo";
        }
    }

    @GetMapping("/adocoes/minhas")
    public String minhasAdocoes(Authentication authentication, Model model) {
        if (authentication == null) {
            return "redirect:/login";
        }
        model.addAttribute("adocoes", adocaoService.listarAdocoesAdotante(authentication.getName()));
        return "adocao/minhas-adocoes";
    }

    @GetMapping("/adocoes/fila")
    public String filaAdocoes(Authentication authentication, Model model) {
        if (authentication == null) {
            return "redirect:/login";
        }
        model.addAttribute("adocoes", adocaoService.listarFilaAdocao(authentication.getName()));
        return "adocao/fila";
    }

    @PostMapping("/adocoes/aprovar")
    public String aprovarAdocao(
            @RequestParam Long adocaoId,
            @RequestParam String disponibilidade,
            Authentication authentication,
            Model model) {
        if (authentication == null) return "redirect:/login";
        try {
            adocaoService.aprovarAdocao(adocaoId, authentication.getName(), disponibilidade);
            model.addAttribute("sucesso", "Pedido de adoção aprovado com sucesso! E-mail enviado informando dias de disponibilidade [RN015].");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("adocoes", adocaoService.listarFilaAdocao(authentication.getName()));
        return "adocao/fila";
    }

    @PostMapping("/adocoes/reprovar")
    public String reprovarAdocao(
            @RequestParam Long adocaoId,
            Authentication authentication,
            Model model) {
        if (authentication == null) return "redirect:/login";
        try {
            adocaoService.reprovarAdocao(adocaoId);
            model.addAttribute("sucesso", "Pedido de adoção reprovado com sucesso! E-mail enviado [RN015].");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("adocoes", adocaoService.listarFilaAdocao(authentication.getName()));
        return "adocao/fila";
    }

    @PostMapping("/adocoes/reprovar-todos")
    public String reprovarTodosDemais(
            @RequestParam Long animalId,
            @RequestParam Long adocaoExcecaoId,
            Authentication authentication,
            Model model) {
        if (authentication == null) return "redirect:/login";
        try {
            adocaoService.reprovarTodosDemais(animalId, adocaoExcecaoId);
            model.addAttribute("sucesso", "Demais pedidos reprovados com sucesso!");
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("adocoes", adocaoService.listarFilaAdocao(authentication.getName()));
        return "adocao/fila";
    }

    @GetMapping("/adocoes/manual")
    public String cadastroManualAdocaoForm(Authentication authentication, Model model) {
        if (authentication == null) return "redirect:/login";
        model.addAttribute("animais", adocaoService.listarAnimaisDisponiveisParaAdocao());
        return "adocao/cadastro-manual";
    }

    @PostMapping("/adocoes/manual")
    public String cadastroManualAdocao(
            Authentication authentication,
            @RequestParam Long animalId,
            @RequestParam String email,
            @RequestParam String nome,
            @RequestParam String cpf,
            @RequestParam LocalDate dataNascimento,
            @RequestParam String telefones,
            @RequestParam String rua,
            @RequestParam String numero,
            @RequestParam(required = false) String complemento,
            @RequestParam String bairro,
            @RequestParam String cidade,
            @RequestParam String estado,
            @RequestParam String cep,
            @RequestParam(required = false) MultipartFile fotoFile,
            @RequestParam String voluntario,
            Model model) {
        if (authentication == null) return "redirect:/login";
        try {
            byte[] fotoBytes = fotoFile != null ? fotoFile.getBytes() : null;
            adocaoService.cadastrarManualEClincharAdocao(animalId, email, nome, cpf, dataNascimento, telefones, rua, numero, complemento, bairro, cidade, estado, cep, fotoBytes, voluntario);
            model.addAttribute(
                "sucesso",
                "Adoção manual cadastrada com sucesso! Senha gerada automaticamente (data de nascimento DD-MM-AAAA) conforme RN014."
            );
        } catch (Exception e) {
            model.addAttribute("erro", e.getMessage());
        }
        model.addAttribute("animais", adocaoService.listarAnimaisDisponiveisParaAdocao());
        return "adocao/cadastro-manual";
    }
}
