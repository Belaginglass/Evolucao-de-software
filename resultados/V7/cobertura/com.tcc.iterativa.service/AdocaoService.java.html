package com.tcc.iterativa.service;

import com.tcc.iterativa.model.Adocao;
import com.tcc.iterativa.model.Animal;
import com.tcc.iterativa.model.Usuario;
import com.tcc.iterativa.repository.AdocaoRepository;
import com.tcc.iterativa.repository.AnimalRepository;
import com.tcc.iterativa.repository.UsuarioRepository;
import org.springframework.stereotype.Service;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.List;
import java.util.Optional;
import java.util.stream.Collectors;

@Service
public class AdocaoService {

    private final AdocaoRepository adocaoRepository;
    private final AnimalRepository animalRepository;
    private final UsuarioRepository usuarioRepository;
    private final UsuarioService usuarioService;

    public AdocaoService(AdocaoRepository adocaoRepository, AnimalRepository animalRepository, UsuarioRepository usuarioRepository, UsuarioService usuarioService) {
        this.adocaoRepository = adocaoRepository;
        this.animalRepository = animalRepository;
        this.usuarioRepository = usuarioRepository;
        this.usuarioService = usuarioService;
    }

    public List<Animal> listarAnimaisDisponiveisParaAdocao() {
        List<Animal> todos = animalRepository.findAll();
        return todos.stream().filter(a -> 
            "Sim".equals(a.getDisponivelAdocao()) &&
            !"Sim".equals(a.getEmTratamento()) &&
            !"Sim".equals(a.getPatrimonioTombado())
        ).collect(Collectors.toList());
    }

    public Optional<Animal> buscarAnimalPorId(Long id) {
        return animalRepository.findById(id);
    }

    public Adocao criarPedidoAdocao(Long animalId, String emailAdotante, boolean termoAceito) throws Exception {
        if (!termoAceito) {
            throw new IllegalArgumentException("O termo de adoção deve ser aceito antes de prosseguir.");
        }
        Optional<Animal> animalOpt = animalRepository.findById(animalId);
        if (animalOpt.isEmpty()) {
            throw new IllegalArgumentException("Animal não encontrado.");
        }
        Animal animal = animalOpt.get();
        if (!"Sim".equals(animal.getDisponivelAdocao()) || "Sim".equals(animal.getEmTratamento()) || "Sim".equals(animal.getPatrimonioTombado())) {
            throw new IllegalArgumentException("Este animal não está disponível para adoção.");
        }

        Optional<Usuario> usuarioOpt = usuarioRepository.findByEmail(emailAdotante);
        if (usuarioOpt.isEmpty()) {
            throw new IllegalArgumentException("Usuário adotante não encontrado.");
        }
        Usuario adotante = usuarioOpt.get();

        // RN007: Gestor não pode adotar seu próprio animal / aprovar próprio pedido

        Adocao adocao = new Adocao(animal, adotante, "AGUARDANDO", LocalDateTime.now());
        return adocaoRepository.save(adocao);
    }

    public List<Adocao> listarAdocoesAdotante(String emailAdotante) {
        Optional<Usuario> usuarioOpt = usuarioRepository.findByEmail(emailAdotante);
        if (usuarioOpt.isPresent()) {
            return adocaoRepository.findByAdotanteOrderByDataPedidoAsc(usuarioOpt.get());
        }
        return List.of();
    }

    public List<Adocao> listarFilaAdocao(String emailGestorLogado) {
        Optional<Usuario> gestorOpt = usuarioRepository.findByEmail(emailGestorLogado);
        List<Adocao> todas = adocaoRepository.findAll();
        
        return todas.stream().filter(a -> {
            // RN007: O sistema não deve listar os pedidos de adoção do próprio gestor
            if (gestorOpt.isPresent() && a.getAdotante().getId().equals(gestorOpt.get().getId())) {
                return false;
            }
            return true;
        }).sorted((a1, a2) -> {
            boolean a1Ag = "AGUARDANDO".equals(a1.getEstado());
            boolean a2Ag = "AGUARDANDO".equals(a2.getEstado());
            if (a1Ag && !a2Ag) return -1;
            if (!a1Ag && a2Ag) return 1;
            return a1.getDataPedido().compareTo(a2.getDataPedido());
        }).collect(Collectors.toList());
    }

    public void aprovarAdocao(Long adocaoId, String emailGestor, String disponibilidade) throws Exception {
        Optional<Adocao> adocaoOpt = adocaoRepository.findById(adocaoId);
        if (adocaoOpt.isEmpty()) {
            throw new IllegalArgumentException("Pedido de adoção não encontrado.");
        }
        Adocao adocao = adocaoOpt.get();
        Optional<Usuario> gestorOpt = usuarioRepository.findByEmail(emailGestor);
        
        // RN007: Gestor não pode aprovar a própria solicitação
        if (gestorOpt.isPresent() && adocao.getAdotante().getId().equals(gestorOpt.get().getId())) {
            throw new IllegalArgumentException("RN007: Um gestor não pode aprovar sua própria solicitação de adoção.");
        }

        adocao.setEstado("AGUARDANDO_VISITA");
        adocaoRepository.save(adocao);
        // RN015: Simular envio de e-mail com disponibilidade
    }

    public void reprovarAdocao(Long adocaoId) {
        Optional<Adocao> adocaoOpt = adocaoRepository.findById(adocaoId);
        if (adocaoOpt.isPresent()) {
            Adocao adocao = adocaoOpt.get();
            adocao.setEstado("REPROVADO");
            adocaoRepository.save(adocao);
            // RN015: Simular envio de e-mail de reprovação
        }
    }

    public void reprovarTodosDemais(Long animalId, Long adocaoExcecaoId) {
        Optional<Animal> animalOpt = animalRepository.findById(animalId);
        if (animalOpt.isPresent()) {
            List<Adocao> adocoes = adocaoRepository.findByAnimal(animalOpt.get());
            for (Adocao a : adocoes) {
                if (!a.getId().equals(adocaoExcecaoId) && "AGUARDANDO".equals(a.getEstado())) {
                    a.setEstado("REPROVADO");
                    adocaoRepository.save(a);
                }
            }
        }
    }

    public void cadastrarManualEClincharAdocao(Long animalId, String email, String nome, String cpf, LocalDate dataNascimento, String telefones, String rua, String numero, String complemento, String bairro, String cidade, String estado, String cep, byte[] fotoBytes, String voluntario) throws Exception {
        // RN014: Senha gerada automaticamente como a data de nascimento no formato dia-mês-ano (ddMMyyyy ou dd-MM-yyyy)
        DateTimeFormatter formatter = DateTimeFormatter.ofPattern("ddMMyyyy");
        String senhaAutomatica = dataNascimento.format(formatter);

        Usuario novoAdotante = usuarioService.cadastrar(email, senhaAutomatica, "ADOTANTE", nome, cpf, dataNascimento, telefones, rua, numero, complemento, bairro, cidade, estado, cep, fotoBytes, voluntario);
        novoAdotante.setSenhaProvisoriaRN014(true);
        usuarioRepository.save(novoAdotante);

        // Criar pedido e já aprovar/vincular para AGUARDANDO_VISITA conforme fluxo manual
        Optional<Animal> animalOpt = animalRepository.findById(animalId);
        if (animalOpt.isPresent()) {
            Animal animal = animalOpt.get();
            Adocao adocao = new Adocao(animal, novoAdotante, "AGUARDANDO_VISITA", LocalDateTime.now());
            adocaoRepository.save(adocao);
        }
    }
}
