package com.tcc.iterativa.model;

import jakarta.persistence.*;
import java.time.LocalDateTime;

@Entity
@Table(name = "adocoes")
public class Adocao {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne
    @JoinColumn(name = "animal_id", nullable = false)
    private Animal animal;

    @ManyToOne
    @JoinColumn(name = "adotante_id", nullable = false)
    private Usuario adotante;

    @Column(nullable = false)
    private String estado; // AGUARDANDO, REPROVADO, AGUARDANDO_VISITA, DESISTENCIA, ADOTADO, DEVOLVIDO, REVOGADO

    @Column(nullable = false)
    private LocalDateTime dataPedido;

    private LocalDateTime dataAdocaoEfetiva;

    @Column(columnDefinition = "TEXT")
    private String mensagemFeedbackGestor;

    @Column(columnDefinition = "TEXT")
    private String respostaFeedbackAdotante;

    @Column(columnDefinition = "TEXT")
    private String fotoRelato;

    private Integer estrelasAvaliacao;

    @Column(columnDefinition = "TEXT")
    private String relatoAdotante;

    private Boolean relatoAprovado; // null = pendente, true = aprovado, false = removido

    @Column(columnDefinition = "TEXT")
    private String motivoRemocaoRelato;

    public Adocao() {}

    public Adocao(Animal animal, Usuario adotante, String estado, LocalDateTime dataPedido) {
        this.animal = animal;
        this.adotante = adotante;
        this.estado = estado;
        this.dataPedido = dataPedido;
    }

    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }

    public Animal getAnimal() { return animal; }
    public void setAnimal(Animal animal) { this.animal = animal; }

    public Usuario getAdotante() { return adotante; }
    public void setAdotante(Usuario adotante) { this.adotante = adotante; }

    public String getEstado() { return estado; }
    public void setEstado(String estado) { this.estado = estado; }

    public LocalDateTime getDataPedido() { return dataPedido; }
    public void setDataPedido(LocalDateTime dataPedido) { this.dataPedido = dataPedido; }

    public LocalDateTime getDataAdocaoEfetiva() { return dataAdocaoEfetiva; }
    public void setDataAdocaoEfetiva(LocalDateTime dataAdocaoEfetiva) { this.dataAdocaoEfetiva = dataAdocaoEfetiva; }

    public String getMensagemFeedbackGestor() { return mensagemFeedbackGestor; }
    public void setMensagemFeedbackGestor(String mensagemFeedbackGestor) { this.mensagemFeedbackGestor = mensagemFeedbackGestor; }

    public String getRespostaFeedbackAdotante() { return respostaFeedbackAdotante; }
    public void setRespostaFeedbackAdotante(String respostaFeedbackAdotante) { this.respostaFeedbackAdotante = respostaFeedbackAdotante; }

    public String getFotoRelato() { return fotoRelato; }
    public void setFotoRelato(String fotoRelato) { this.fotoRelato = fotoRelato; }

    public Integer getEstrelasAvaliacao() { return estrelasAvaliacao; }
    public void setEstrelasAvaliacao(Integer estrelasAvaliacao) { this.estrelasAvaliacao = estrelasAvaliacao; }

    public String getRelatoAdotante() { return relatoAdotante; }
    public void setRelatoAdotante(String relatoAdotante) { this.relatoAdotante = relatoAdotante; }

    public Boolean getRelatoAprovado() { return relatoAprovado; }
    public void setRelatoAprovado(Boolean relatoAprovado) { this.relatoAprovado = relatoAprovado; }

    public String getMotivoRemocaoRelato() { return motivoRemocaoRelato; }
    public void setMotivoRemocaoRelato(String motivoRemocaoRelato) { this.motivoRemocaoRelato = motivoRemocaoRelato; }
}
