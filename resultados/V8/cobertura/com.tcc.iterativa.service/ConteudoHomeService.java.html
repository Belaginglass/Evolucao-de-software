package com.tcc.iterativa.service;

import com.tcc.iterativa.model.ConteudoHome;
import com.tcc.iterativa.repository.ConteudoHomeRepository;
import org.springframework.stereotype.Service;

import java.util.Collections;
import java.util.List;
import java.util.stream.Collectors;

@Service
public class ConteudoHomeService {

    private final ConteudoHomeRepository repository;

    public ConteudoHomeService(ConteudoHomeRepository repository) {
        this.repository = repository;
    }

    public String obterSobreNos() {
        List<ConteudoHome> list = repository.findByTipo("SOBRE");
        if (!list.isEmpty()) {
            return list.get(0).getDescricao();
        }
        return "DogTech - Sistema de Gerenciamento de Adoção.";
    }

    public void salvarSobreNos(String texto) {
        List<ConteudoHome> list = repository.findByTipo("SOBRE");
        ConteudoHome sobre;
        if (!list.isEmpty()) {
            sobre = list.get(0);
        } else {
            sobre = new ConteudoHome();
            sobre.setTipo("SOBRE");
        }
        sobre.setDescricao(texto);
        repository.save(sobre);
    }

    public List<ConteudoHome> listarServicos() {
        return repository.findByTipo("SERVICO");
    }

    public List<ConteudoHome> listarCampanhas() {
        return repository.findByTipo("CAMPANHA");
    }

    public List<ConteudoHome> listarDoacoes() {
        return repository.findByTipo("DOACAO");
    }

    public void salvarItem(String tipo, String titulo, String descricao, String valor) {
        if (titulo == null || titulo.trim().isEmpty()) {
            throw new IllegalArgumentException("O título é obrigatório.");
        }
        ConteudoHome item = new ConteudoHome(tipo, titulo, descricao, valor);
        repository.save(item);
    }

    public void deletarItem(Long id) {
        repository.deleteById(id);
    }

    public List<ConteudoHome> selecionarAleatorio(List<ConteudoHome> lista, int max) {
        if (lista == null || lista.isEmpty()) return Collections.emptyList();
        List<ConteudoHome> copia = lista.stream().collect(Collectors.toList());
        Collections.shuffle(copia);
        return copia.stream().limit(max).collect(Collectors.toList());
    }
}
